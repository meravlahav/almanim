name: Weekly link check

on:
  schedule:
    - cron: '17 5 * * 0'   # every Sunday 05:17 UTC (08:17 Israel time)
  workflow_dispatch:

permissions:
  contents: read

jobs:
  links:
    name: Check every external link in the content
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - run: node scripts/check-links.js
